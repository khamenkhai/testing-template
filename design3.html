  %% =========================
  %% CORE TABLES
  %% =========================

  ORGANIZATIONS {
    int id PK
    string name
    string subdomain
    timestamp created_at
  }

  BRANCHES {
    int id PK
    int organization_id FK
    string name
    string address
    string business_type
    timestamp created_at
  }

  USERS {
    int id PK
    int organization_id FK
    string name
    string email
    string password_hash
    string role
    timestamp created_at
  }

  TABLES {
    int id PK
    int branch_id FK
    string table_number
    int seating_capacity
    string status
  }


  %% =========================
  %% PRODUCT SYSTEM (LOYVERSE CORE)
  %% =========================

  PRODUCT_CATEGORIES {
    int id PK
    string name
  }

  PRODUCTS {
    int id PK
    int category_id FK
    string name
    string description
    boolean is_active
  }

  BRANCH_PRODUCTS {
    int id PK
    int branch_id FK
    int product_id FK
    boolean is_available
  }

  PRODUCT_VARIANTS {
    int id PK
    int branch_product_id FK
    string name
    string sku
    string barcode
    decimal price
    int uom_id FK
    decimal base_unit_qty
    boolean track_inventory
  }


  %% =========================
  %% UNITS OF MEASURE (RETAIL REQUIRED)
  %% =========================

  UOM {

    string name

    boolean is_base_unit
  }

  %% ORDER SYSTEM (UNIFIED)
  %% =========================

  ORDERS {

    int branch_id FK

    string order_channel

    timestamp created_at

  ORDER_ITEMS {
    int id PK
    int order_id FK

    text notes
  }

    int id PK
    int order_item_id FK
    int modifier_id FK
    decimal extra_price
  }

  MODIFIERS {
    int id PK
    int product_id FK
    string name
    decimal price
  }

  %% INVENTORY SYSTEM (RETAIL CORE)
